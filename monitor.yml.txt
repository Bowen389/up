name: 悠悠求购监控

on:
  workflow_dispatch:            # Actions 页面手动 Run workflow
  schedule:
    - cron: "7 9 * * *"         # 每天北京时间 17:07（UTC 9:07）

permissions:
  contents: write

concurrency:
  group: yyyp-monitor
  cancel-in-progress: false

jobs:
  run:
    runs-on: ubuntu-latest
    timeout-minutes: 240
    env:
      YP_RAW_HEADERS: ${{ secrets.YP_RAW_HEADERS }}       # 推荐：整段请求头
      YP_AUTHORIZATION: ${{ secrets.YP_AUTHORIZATION }}   # 或分别填 4 个（二选一）
      YP_DEVICE_ID: ${{ secrets.YP_DEVICE_ID }}
      YP_DEVICE_UK: ${{ secrets.YP_DEVICE_UK }}
      YP_UK: ${{ secrets.YP_UK }}
      TZ: Asia/Shanghai
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: 检查文件
        run: |
          echo "仓库根目录内容："; ls -la
          test -f monitor.py || { echo "❌ 根目录没有 monitor.py"; exit 1; }
          test -f items.txt  || { echo "❌ 根目录没有 items.txt"; exit 1; }
          grep -q "_load_cfg" monitor.py || { echo "❌ monitor.py 是旧版本，请替换为新版（需包含 _load_cfg）"; exit 1; }
          mkdir -p data
          echo "✅ 文件检查通过，监控商品数：$(grep -c '^[0-9]' items.txt)"

      - run: pip install requests

      - name: 接口预检
        run: |
          python - <<'PY'
          import sys, monitor
          monitor.check_token()
          d = monitor.post('/api/youpin/bff/trade/purchase/order/getTemplatePurchaseOrderListPC',
                           {'templateId': '43397', 'pageIndex': 1, 'pageSize': 10})
          ok1 = d.get('code') == 0
          d2 = monitor.post('/api/homepage/pc/goods/market/querySaleTemplate',
                            {'gameId': '730', 'listType': '10', 'keyWords': 'AWP', 'pageIndex': 1,
                             'pageSize': 10, 'listSortType': 0, 'sortType': 0})
          ok2 = d2.get('Code') == 0
          print('求购接口', '✅' if ok1 else f'❌ {d}')
          print('搜索接口', '✅' if ok2 else f'❌ {d2}')
          if not (ok1 and ok2):
              sys.exit('接口不可用（可能是海外 IP 被拦截或 token 失效）')
          PY

      - name: 运行监控
        run: python -u monitor.py

      - name: 提交数据和报告
        if: always()
        run: |
          mkdir -p data
          git config user.name  "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"
          git add -A data/ || true
          git diff --cached --quiet || git commit -m "data: $(date +'%F %H:%M')"
          git push || true

      - name: 上传报告附件
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: report
          path: data/report.html
          if-no-files-found: ignore
